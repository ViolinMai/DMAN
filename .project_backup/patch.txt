    async function doAction(action, gid) {
      await apiRequest("/api/action", "POST", { action, gid });

    // --- Folder Picker Modal Logic ---
    let pickerSubpath = "";

    function openFolderPickerModal() {
      pickerSubpath = "";
      document.getElementById("folderPickerModal").style.display = "flex";
      loadPickerFolders("");
    }

    function closeFolderPickerModal() {
      document.getElementById("folderPickerModal").style.display = "none";
    }

    async function loadPickerFolders(subpath = "") {
      pickerSubpath = subpath;
      document.getElementById("pickerCurrentPath").innerText = subpath || "root";
      document.getElementById("pickerUpBtn").style.display = subpath ? "inline-block" : "none";

      const res = await apiRequest("/api/fs/list", "POST", { subpath });
      const container = document.getElementById("pickerItemsContainer");

      if (!res || res.status !== "ok") {
        container.innerHTML = '<div style="color: var(--text-muted); font-size: 0.85rem; padding: 10px;">Cannot read folder.</div>';
        return;
      }

      const folders = (res.items || []).filter(item => item.isDir);
      if (folders.length === 0) {
        container.innerHTML = '<div style="color: var(--text-muted); font-size: 0.85rem; padding: 10px;">No subfolders inside. Click "Choose Current Folder" below to select this path.</div>';
        return;
      }

      container.innerHTML = folders.map(folder => `
        <div class="explorer-item" style="cursor: pointer;" onclick="navigatePickerInto('${escapeHtml(folder.name)}')">
          <div style="display: flex; align-items: center; gap: 8px;">
            <span>📁</span>
            <span style="font-weight: bold;">${escapeHtml(folder.name)}</span>
          </div>
          <span style="color: var(--accent-blue); font-size: 0.8rem;">Open ➔</span>
        </div>
      `).join("");
    }

    function navigatePickerInto(name) {
      const next = pickerSubpath ? `${pickerSubpath}/${name}` : name;
      loadPickerFolders(next);
    }

    function navigatePickerUp() {
      if (!pickerSubpath) return;
      const parts = pickerSubpath.split("/").filter(Boolean);
      parts.pop();
      loadPickerFolders(parts.join("/"));
    }

    async function confirmPickerFolder() {
      const stateRes = await apiRequest("/api/status");
      const baseDir = (stateRes && stateRes.data && stateRes.data.defaultSaveDir) ? stateRes.data.defaultSaveDir.replace(/[\/\\]+$/, "") : "";
      const chosen = pickerSubpath ? `${baseDir}/${pickerSubpath}` : baseDir;
      document.getElementById("targetDirInput").value = chosen;
      localStorage.setItem("dman_saved_dir", chosen);
      closeFolderPickerModal();
    }
