  <script>
    // Load stored values automatically from localStorage
    let host = (localStorage.getItem("dman_host") || "inclusive-legislative-steady-tide.trycloudflare.com").trim();
    let token = (localStorage.getItem("dman_token") || "dman_remote_2026").trim();
    let savedDir = localStorage.getItem("dman_saved_dir") || "";

    document.getElementById("serverHost").value = host;
    document.getElementById("authToken").value = token;
    document.getElementById("targetDirInput").value = savedDir;

    function buildBaseUrl(raw) {
      let cleaned = raw.trim();
      if (!cleaned) return "http://127.0.0.1:6801";

      // If full protocol already supplied
      if (cleaned.startsWith("http://") || cleaned.startsWith("https://")) {
        return cleaned.replace(/\/+$/, "");
      }

      // If it is a Cloudflare Tunnel or domain name -> use https without custom port
      if (cleaned.includes("trycloudflare.com") || cleaned.includes(".")) {
        return `https://${cleaned}`.replace(/\/+$/, "");
      }

      // Fallback for local IP / localhost
      return `http://${cleaned}:6801`.replace(/\/+$/, "");
    }

    function saveAndConnect() {
      host = document.getElementById("serverHost").value.trim();
      token = document.getElementById("authToken").value.trim();
      localStorage.setItem("dman_host", host);
      localStorage.setItem("dman_token", token);
      fetchStatus();
    }

    async function apiRequest(endpoint, method = "GET", body = null) {
      const baseUrl = buildBaseUrl(host);
      const url = `${baseUrl}${endpoint}`;
      const headers = {
        "Content-Type": "application/json",
        "X-DMAN-Token": token
      };

      try {
        const controller = new AbortController();
        const timeoutId = setTimeout(() => controller.abort(), 6000);
        const resp = await fetch(url, {
          method,
          headers,
          body: body ? JSON.stringify(body) : null,
          signal: controller.signal
        });
        clearTimeout(timeoutId);
        if (!resp.ok) return null;
        return await resp.json();
      } catch (e) {
        return null;
      }
    }

    async function fetchStatus() {
      const res = await apiRequest("/api/status");
      const dot = document.getElementById("statusDot");
      const stText = document.getElementById("statusText");

      if (!res || res.status !== "ok") {
        dot.className = "dot";
        stText.innerText = "Offline";
        return;
      }

      dot.className = "dot online";
      stText.innerText = "Connected";

      const d = res.data;
      document.getElementById("statSpeed").innerText = d.globalSpeedStr || "0 B/s";
      document.getElementById("statActive").innerText = d.activeDownloads || 0;
      document.getElementById("statDisk").innerText = d.freeSpaceStr || "-";
      document.getElementById("statAD").innerText = d.alldebridStatus || "Disabled";

      if (!document.getElementById("targetDirInput").value && d.defaultSaveDir) {
        document.getElementById("targetDirInput").value = d.defaultSaveDir;
      }

      renderTable(d.downloads || []);
    }

    function renderTable(items) {
      const tbody = document.getElementById("downloadsTableBody");
      if (!items || items.length === 0) {
        tbody.innerHTML = '<tr><td colspan="7" style="text-align: center; color: var(--text-muted); padding: 30px;">Queue is empty.</td></tr>';
        return;
      }

      tbody.innerHTML = items.map(it => {
        const isDone = it.status === "complete";
        const fillClass = isDone ? "progress-fill done" : "progress-fill";
        return `
          <tr>
            <td style="max-width: 260px; overflow: hidden; text-overflow: ellipsis; white-space: nowrap;" title="${it.name}">
              ${it.name}
            </td>
            <td>${it.sizeStr || "-"}</td>
            <td>${it.statusDisplay || it.status}</td>
            <td style="color: var(--accent-blue);">${it.speedStr || "-"}</td>
            <td>${it.etaStr || "~"}</td>
            <td class="progress-cell">
              <div class="progress-bar">
                <div class="${fillClass}" style="width: ${it.progress || 0}%;"></div>
                <div class="progress-text">${(it.progress || 0).toFixed(0)}%</div>
              </div>
            </td>
            <td class="actions-cell">
              ${it.status === "active" ? `<button onclick="doAction('pause', '${it.gid}')">Pause</button>` : ""}
              ${it.status === "paused" ? `<button onclick="doAction('resume', '${it.gid}')">Resume</button>` : ""}
              <button onclick="doAction('remove', '${it.gid}')" style="color: var(--status-error);">✕</button>
            </td>
          </tr>
        `;
      }).join("");
    }

    async function submitLinks() {
      const text = document.getElementById("linksInput").value.trim();
      if (!text) return;

      const saveDir = document.getElementById("targetDirInput").value.trim();
      const subfolder = document.getElementById("subfolderInput").value.trim();
      const useAD = document.getElementById("useADCheck").checked;

      if (saveDir) {
        localStorage.setItem("dman_saved_dir", saveDir);
      }

      await apiRequest("/api/add", "POST", { links: text, saveDir, subfolder, useAlldebrid: useAD });
      document.getElementById("linksInput").value = "";
      document.getElementById("subfolderInput").value = "";
      fetchStatus();
    }

    async function doAction(action, gid) {
      await apiRequest("/api/action", "POST", { action, gid });
      fetchStatus();
    }

    // Auto-refresh every 2 seconds
    setInterval(fetchStatus, 2000);
    fetchStatus();
  </script>
